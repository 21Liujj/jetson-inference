cmake_minimum_required(VERSION 3.16)
project(edgevision LANGUAGES CXX)
include(CTest)
find_package(Threads REQUIRED)
option(EDGEVISION_ENABLE_V4L2 "Build the real Linux V4L2 capture backend" OFF)
option(EDGEVISION_ENABLE_SANITIZERS "Enable ASan/UBSan for ownership tests" OFF)
option(EDGEVISION_ENABLE_UPSTREAM_INFERENCE "Link the repaired upstream DetectNet baseline" OFF)
option(EDGEVISION_BUILD_GPU_ACCEPTANCE "Build explicit GPU delay-injection acceptance tools" OFF)
option(EDGEVISION_ENABLE_CUDA_PIPELINE "Build the numerically checked CUDA preprocessing/pipeline" OFF)
option(EDGEVISION_ENABLE_AGENT_BACKEND "Build bounded snapshots and the owned local VLM process adapter" OFF)
set(EDGEVISION_UPSTREAM_PREFIX "" CACHE PATH "Own upstream build's aarch64 directory (include/ and lib/)")

add_library(edgevision_core src/frame/frame.cc src/frame/preview_cache.cc src/frame/snapshot_store.cc src/task_manager.cc src/control_router.cc src/health_monitor.cc)
target_include_directories(edgevision_core PUBLIC include)
target_compile_features(edgevision_core PUBLIC cxx_std_17)
target_link_libraries(edgevision_core PUBLIC Threads::Threads)
if(EDGEVISION_ENABLE_SANITIZERS)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" OR WIN32)
    message(FATAL_ERROR "ASan/UBSan preset requires a supported Linux GCC/Clang environment")
  endif()
  target_compile_options(edgevision_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(edgevision_core PUBLIC -fsanitize=address,undefined)
endif()
if(EDGEVISION_ENABLE_V4L2)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The real V4L2 backend requires Linux; core-only builds use OFF.")
  endif()
  add_library(edgevision_capture src/camera/v4l2_capture.cc src/camera/v4l2_backend.cc)
  target_link_libraries(edgevision_capture PUBLIC edgevision_core)
  target_compile_options(edgevision_capture PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(edgevision_capture_probe tools/capture_probe.cpp)
  target_link_libraries(edgevision_capture_probe PRIVATE edgevision_capture)
  add_executable(edgevision_recovery_probe tools/capture_recovery_probe.cpp)
  target_link_libraries(edgevision_recovery_probe PRIVATE edgevision_capture)
endif()
if(EDGEVISION_ENABLE_UPSTREAM_INFERENCE)
  if(NOT EDGEVISION_ENABLE_V4L2 OR NOT IS_DIRECTORY "${EDGEVISION_UPSTREAM_PREFIX}/include/jetson-inference")
    message(FATAL_ERROR "Upstream inference requires V4L2 and an explicit verified upstream build prefix")
  endif()
  find_package(CUDAToolkit REQUIRED)
  find_package(OpenCV REQUIRED COMPONENTS core imgproc)
  find_library(EDGEVISION_INFERENCE_LIBRARY NAMES jetson-inference PATHS "${EDGEVISION_UPSTREAM_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
  find_library(EDGEVISION_UTILS_LIBRARY NAMES jetson-utils PATHS "${EDGEVISION_UPSTREAM_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
  add_library(edgevision_inference src/inference/upstream_detector.cc src/inference/upstream_executor.cc src/inference/upstream_yolo.cc)
  target_include_directories(edgevision_inference PRIVATE "${EDGEVISION_UPSTREAM_PREFIX}/include")
  target_link_libraries(edgevision_inference PUBLIC edgevision_capture CUDA::cudart PRIVATE "${EDGEVISION_INFERENCE_LIBRARY}" "${EDGEVISION_UTILS_LIBRARY}")
  target_compile_options(edgevision_inference PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(edgevision_inference_probe tools/capture_inference_probe.cpp)
  target_link_libraries(edgevision_inference_probe PRIVATE edgevision_capture edgevision_inference)
  add_executable(edgevision_yolo_tensor_probe tools/yolo_tensor_probe.cpp)
  target_link_libraries(edgevision_yolo_tensor_probe PRIVATE edgevision_inference)
  add_library(edgevision_yolo_preprocess src/inference/yolo_preprocess.cc)
  target_include_directories(edgevision_yolo_preprocess PUBLIC include PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_compile_features(edgevision_yolo_preprocess PUBLIC cxx_std_17)
  target_link_libraries(edgevision_yolo_preprocess PRIVATE ${OpenCV_LIBS})
  target_link_libraries(edgevision_inference PRIVATE edgevision_yolo_preprocess)
  add_executable(edgevision_yolo_preprocess_probe tools/yolo_preprocess_probe.cpp)
  target_link_libraries(edgevision_yolo_preprocess_probe PRIVATE edgevision_yolo_preprocess)
endif()
if(EDGEVISION_ENABLE_CUDA_PIPELINE)
  if(NOT EDGEVISION_ENABLE_UPSTREAM_INFERENCE)
    message(FATAL_ERROR "CUDA pipeline requires the explicit upstream inference build")
  endif()
  enable_language(CUDA)
  target_compile_definitions(edgevision_inference PUBLIC EDGEVISION_NVTX=1 PRIVATE EDGEVISION_HAS_CUDA_PIPELINE=1)
  target_link_libraries(edgevision_inference PRIVATE ${CMAKE_DL_LIBS})
  add_library(edgevision_cuda_preprocess STATIC src/inference/cuda_letterbox.cu)
  target_include_directories(edgevision_cuda_preprocess PUBLIC include)
  set_target_properties(edgevision_cuda_preprocess PROPERTIES CUDA_STANDARD 17 CUDA_ARCHITECTURES 87 CUDA_RUNTIME_LIBRARY Shared)
  target_link_libraries(edgevision_cuda_preprocess PUBLIC CUDA::cudart)
  target_sources(edgevision_inference PRIVATE src/inference/gpu_yolo_pipeline.cc)
  target_link_libraries(edgevision_inference PRIVATE edgevision_cuda_preprocess)
  add_executable(edgevision_cuda_preprocess_probe tools/cuda_preprocess_probe.cpp)
  target_link_libraries(edgevision_cuda_preprocess_probe PRIVATE edgevision_cuda_preprocess)
  set_target_properties(edgevision_cuda_preprocess_probe PROPERTIES CUDA_RUNTIME_LIBRARY Shared)
  add_executable(edgevision_gpu_pipeline_probe tools/gpu_pipeline_probe.cpp)
  target_link_libraries(edgevision_gpu_pipeline_probe PRIVATE edgevision_inference)
endif()
if(EDGEVISION_ENABLE_AGENT_BACKEND)
  if(NOT EDGEVISION_ENABLE_UPSTREAM_INFERENCE)
    message(FATAL_ERROR "Agent backend requires the shared physical inference executor")
  endif()
  find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
  find_package(nlohmann_json REQUIRED)
  target_sources(edgevision_inference PRIVATE src/agent/agent_tools.cc src/agent/vlm_process.cc)
  target_compile_definitions(edgevision_inference PRIVATE EDGEVISION_HAS_AGENT_BACKEND=1)
  target_include_directories(edgevision_inference PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(edgevision_inference PRIVATE ${OpenCV_LIBS} nlohmann_json::nlohmann_json)
endif()
if(EDGEVISION_BUILD_GPU_ACCEPTANCE)
  if(NOT EDGEVISION_ENABLE_UPSTREAM_INFERENCE)
    message(FATAL_ERROR "GPU acceptance requires the real upstream inference backend")
  endif()
  enable_language(CUDA)
  add_library(edgevision_gpu_acceptance_delay STATIC tools/gpu_acceptance_delay.cu)
  set_target_properties(edgevision_gpu_acceptance_delay PROPERTIES CUDA_STANDARD 14 CUDA_ARCHITECTURES 87 CUDA_RUNTIME_LIBRARY Shared)
  target_link_libraries(edgevision_gpu_acceptance_delay PUBLIC CUDA::cudart)
  add_executable(edgevision_executor_probe tools/executor_probe.cpp)
  target_link_libraries(edgevision_executor_probe PRIVATE edgevision_inference edgevision_gpu_acceptance_delay)
  set_target_properties(edgevision_executor_probe PROPERTIES CUDA_RUNTIME_LIBRARY Shared)
  add_executable(edgevision_executor_recovery_probe tools/executor_recovery_probe.cpp)
  target_link_libraries(edgevision_executor_recovery_probe PRIVATE edgevision_inference edgevision_gpu_acceptance_delay)
  set_target_properties(edgevision_executor_recovery_probe PROPERTIES CUDA_RUNTIME_LIBRARY Shared)
endif()
if(BUILD_TESTING)
  find_package(Python3 3.10 QUIET COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME agent_contracts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_agent_contracts.py")
    set_tests_properties(agent_contracts PROPERTIES TIMEOUT 30)
  endif()
  foreach(name bounded_queue frame_lease frame_queue preview_cache snapshot_store task_manager control_router capture_recovery health_monitor image_layout execution_router yolo_postprocess)
    add_executable(test_${name} tests/test_${name}.cc)
    target_link_libraries(test_${name} PRIVATE edgevision_core)
    if(MSVC)
      target_compile_options(test_${name} PRIVATE /UNDEBUG)
    else()
      target_compile_options(test_${name} PRIVATE -UNDEBUG)
    endif()
    add_test(NAME ${name} COMMAND test_${name})
    set_tests_properties(${name} PROPERTIES TIMEOUT 30)
  endforeach()
  if(EDGEVISION_ENABLE_V4L2)
    foreach(name v4l2_capture v4l2_backend)
      add_executable(test_${name} tests/test_${name}.cc)
      target_link_libraries(test_${name} PRIVATE edgevision_capture)
      target_compile_options(test_${name} PRIVATE -UNDEBUG)
      add_test(NAME ${name} COMMAND test_${name})
      set_tests_properties(${name} PROPERTIES TIMEOUT 30)
    endforeach()
  endif()
endif()
