cmake_minimum_required(VERSION 3.16)
project(edgevision LANGUAGES CXX)
include(CTest)
find_package(Threads REQUIRED)
option(EDGEVISION_ENABLE_V4L2 "Build the real Linux V4L2 capture backend" OFF)
option(EDGEVISION_ENABLE_SANITIZERS "Enable ASan/UBSan for ownership tests" OFF)
option(EDGEVISION_ENABLE_UPSTREAM_INFERENCE "Link the repaired upstream DetectNet baseline" OFF)
set(EDGEVISION_UPSTREAM_PREFIX "" CACHE PATH "Own upstream build's aarch64 directory (include/ and lib/)")

add_library(edgevision_core src/frame/frame.cc src/task_manager.cc src/control_router.cc src/health_monitor.cc)
target_include_directories(edgevision_core PUBLIC include)
target_compile_features(edgevision_core PUBLIC cxx_std_17)
target_link_libraries(edgevision_core PUBLIC Threads::Threads)
if(EDGEVISION_ENABLE_SANITIZERS)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" OR WIN32)
    message(FATAL_ERROR "ASan/UBSan preset requires a supported Linux GCC/Clang environment")
  endif()
  target_compile_options(edgevision_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(edgevision_core PUBLIC -fsanitize=address,undefined)
endif()
if(EDGEVISION_ENABLE_V4L2)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The real V4L2 backend requires Linux; core-only builds use OFF.")
  endif()
  add_library(edgevision_capture src/camera/v4l2_capture.cc src/camera/v4l2_backend.cc)
  target_link_libraries(edgevision_capture PUBLIC edgevision_core)
  target_compile_options(edgevision_capture PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(edgevision_capture_probe tools/capture_probe.cpp)
  target_link_libraries(edgevision_capture_probe PRIVATE edgevision_capture)
  add_executable(edgevision_recovery_probe tools/capture_recovery_probe.cpp)
  target_link_libraries(edgevision_recovery_probe PRIVATE edgevision_capture)
endif()
if(EDGEVISION_ENABLE_UPSTREAM_INFERENCE)
  if(NOT EDGEVISION_ENABLE_V4L2 OR NOT IS_DIRECTORY "${EDGEVISION_UPSTREAM_PREFIX}/include/jetson-inference")
    message(FATAL_ERROR "Upstream inference requires V4L2 and an explicit verified upstream build prefix")
  endif()
  find_package(CUDAToolkit REQUIRED)
  find_library(EDGEVISION_INFERENCE_LIBRARY NAMES jetson-inference PATHS "${EDGEVISION_UPSTREAM_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
  find_library(EDGEVISION_UTILS_LIBRARY NAMES jetson-utils PATHS "${EDGEVISION_UPSTREAM_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
  add_library(edgevision_inference src/inference/upstream_detector.cc)
  target_include_directories(edgevision_inference PRIVATE "${EDGEVISION_UPSTREAM_PREFIX}/include")
  target_link_libraries(edgevision_inference PUBLIC edgevision_core PRIVATE "${EDGEVISION_INFERENCE_LIBRARY}" "${EDGEVISION_UTILS_LIBRARY}" CUDA::cudart)
  target_compile_options(edgevision_inference PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(edgevision_inference_probe tools/capture_inference_probe.cpp)
  target_link_libraries(edgevision_inference_probe PRIVATE edgevision_capture edgevision_inference)
endif()
if(BUILD_TESTING)
  foreach(name bounded_queue frame_lease frame_queue task_manager control_router capture_recovery health_monitor image_layout)
    add_executable(test_${name} tests/test_${name}.cc)
    target_link_libraries(test_${name} PRIVATE edgevision_core)
    if(MSVC)
      target_compile_options(test_${name} PRIVATE /UNDEBUG)
    else()
      target_compile_options(test_${name} PRIVATE -UNDEBUG)
    endif()
    add_test(NAME ${name} COMMAND test_${name})
    set_tests_properties(${name} PROPERTIES TIMEOUT 30)
  endforeach()
  if(EDGEVISION_ENABLE_V4L2)
    foreach(name v4l2_capture v4l2_backend)
      add_executable(test_${name} tests/test_${name}.cc)
      target_link_libraries(test_${name} PRIVATE edgevision_capture)
      target_compile_options(test_${name} PRIVATE -UNDEBUG)
      add_test(NAME ${name} COMMAND test_${name})
      set_tests_properties(${name} PROPERTIES TIMEOUT 30)
    endforeach()
  endif()
endif()
