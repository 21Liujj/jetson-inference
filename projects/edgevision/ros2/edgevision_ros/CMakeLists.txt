cmake_minimum_required(VERSION 3.16)
project(edgevision_ros LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(vision_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(edgevision_interfaces REQUIRED)
find_package(nlohmann_json REQUIRED)
option(EDGEVISION_ROS_BUILD_GPU_ACCEPTANCE "Build separately named 200ms GPU injection node" OFF)
if(EDGEVISION_ROS_BUILD_GPU_ACCEPTANCE)
  set(EDGEVISION_BUILD_GPU_ACCEPTANCE ON CACHE BOOL "" FORCE)
endif()
set(EDGEVISION_ENABLE_V4L2 ON CACHE BOOL "" FORCE)
set(EDGEVISION_ENABLE_UPSTREAM_INFERENCE ON CACHE BOOL "" FORCE)
set(EDGEVISION_ENABLE_CUDA_PIPELINE ON CACHE BOOL "" FORCE)
set(EDGEVISION_ENABLE_AGENT_BACKEND ON CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../.." edgevision-core)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
add_executable(edgevision_node src/edgevision_node.cpp)
target_compile_features(edgevision_node PRIVATE cxx_std_17)
target_link_libraries(edgevision_node edgevision_inference nlohmann_json::nlohmann_json ${OpenCV_LIBS})
target_include_directories(edgevision_node PRIVATE ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(edgevision_node rclcpp rclcpp_action sensor_msgs vision_msgs diagnostic_msgs edgevision_interfaces)
target_compile_definitions(edgevision_node PRIVATE EDGEVISION_VERIFIED_UPSTREAM_PREFIX="${EDGEVISION_UPSTREAM_PREFIX}")
set_target_properties(edgevision_node PROPERTIES INSTALL_RPATH "${EDGEVISION_UPSTREAM_PREFIX}/lib" INSTALL_RPATH_USE_LINK_PATH TRUE)
install(TARGETS edgevision_node DESTINATION lib/${PROJECT_NAME})
if(EDGEVISION_ROS_BUILD_GPU_ACCEPTANCE)
  add_executable(edgevision_gpu_acceptance_node src/edgevision_node.cpp)
  target_compile_features(edgevision_gpu_acceptance_node PRIVATE cxx_std_17)
  target_compile_definitions(edgevision_gpu_acceptance_node PRIVATE EDGEVISION_ROS_GPU_ACCEPTANCE=1)
  target_include_directories(edgevision_gpu_acceptance_node PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(edgevision_gpu_acceptance_node edgevision_inference edgevision_gpu_acceptance_delay nlohmann_json::nlohmann_json ${OpenCV_LIBS})
  ament_target_dependencies(edgevision_gpu_acceptance_node rclcpp rclcpp_action sensor_msgs vision_msgs diagnostic_msgs edgevision_interfaces)
  target_compile_definitions(edgevision_gpu_acceptance_node PRIVATE EDGEVISION_VERIFIED_UPSTREAM_PREFIX="${EDGEVISION_UPSTREAM_PREFIX}")
  set_target_properties(edgevision_gpu_acceptance_node PROPERTIES INSTALL_RPATH "${EDGEVISION_UPSTREAM_PREFIX}/lib" INSTALL_RPATH_USE_LINK_PATH TRUE)
  install(TARGETS edgevision_gpu_acceptance_node DESTINATION lib/${PROJECT_NAME})
endif()
ament_package()
